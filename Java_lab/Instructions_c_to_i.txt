# EXPERIMENT 3 – Basic programming

# Problem with nextInt()
In Java's Scanner class, `nextInt()` (or similar primitive methods like `nextDouble()`) reads only the number token and leaves the trailing newline (`\n`) character from the Enter key in the input buffer. It happens as Scanner methods like `nextInt()` parse until a delimiter (default: whitespace including newlines) but do not consume it. The next `nextLine()` immediately grabs that leftover empty line, skipping intended input.

# 3.1 cBuffered.java
// Demonstrates how to use `BufferedReader` and `InputStreamReader` for input handling.
import java.io.*; 
public class cBuffered {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(
                new InputStreamReader(System.in)
        );
        System.out.print("Enter your age: ");
        int age = Integer.parseInt(br.readLine());
        System.out.print("Enter your name: ");
        String name = br.readLine();
        // Output
        System.out.println("Name: " + name);
        System.out.println("Age: " + age);
    }
}

# 3.2 cScanner.java
// Demonstrates the `nextInt()` newline consumption issue and how to resolve it using an extra `nextLine()`.
import java.util.Scanner;
public class cScanner {
    public static void main(String[] args) {
        try (Scanner obj = new Scanner(System.in)) {
            System.out.print("Enter your age: ");
            int age = obj.nextInt();
            System.out.print("Enter your name: ");
            String name = obj.nextLine(); // Consume the newline character left by nextInt()
            System.out.println("Age: " + age);
            System.out.println("Name: " + name);
        }
    }
}

# EXPERIMENT 4 – Inheritance

# 4.1 dp1.java
// Parent class defining the base properties and methods for inheritance.
// inheritance is a fundamental concept that allows a new class (child class) to inherit properties and behaviors (methods) from an existing class.
class dp1 {
    void show() {
        System.out.println("This is the Parent class");
    }
}

# 4.2 dc1.java
// Demonstrates Single Inheritance where `dc1` extends the parent class `dp1`.
class dc1 extends dp1 {
    void display() {
        System.out.println("This is class c1");
    }
    public static void main(String[] args) {
        dc1 obj = new dc1();
        obj.display();
        obj.show();
    }
}

# 4.3 dc2.java
// Demonstrates Multilevel Inheritance where `dc2` extends `dc1`, which extends `dp1`.
class dc2 extends dc1{
    void Showcase() {
        System.out.println("This is class c2");
    }
    public static void main(String[] args) {
        dc2 obj = new dc2();
        obj.Showcase();
        obj.display();
        obj.show();
    }
}

# 4.4 dp2.java
// Demonstrates Hierarchical Inheritance where `dp2` also extends the base `dp1` class.

    void jolly() {
        System.out.println("This is class c2");
    }
    public static void main(String[] args) {
        dp2 obj = new dp2();

    }
}

# EXPERIMENT 5 – Abstract Classes & Interfaces

# 5.1 eabstract.java
// Demonstrates the concept of abstract classes where an abstract method is declared without a body,
// and the subclass provides its implementation. Also shows usage of concrete methods in abstract classes.
abstract class EntityBase {
    abstract void show();// Abstract method (no body)
    int a = 10000;
    // Regular concrete method
    void display() {
        System.out.println("Hello from abstract class EntityBase");
    }
}

public class eabstract extends EntityBase {
    @Override //we must override the abstract method in the subclass because it has no body in the abstract class
    void show() {
        a = a + 1; // Abstract class variables are not final by default
        System.out.println("This is class abstract. Value of inherited variable 'a' is " + a);
    }
    public static void main(String[] args) {
        System.out.println("--- Abstract Class ---");
        //EntityBase e = new EntityBase(); --error
        eabstract obj = new eabstract(); // Creating an instance of the subclass to access the abstract method
        obj.show();
        obj.display(); // Calling inherited concrete method
    }
}

# 5.2 eI1.java
// interface inheritance (Interface extending another Interface)
interface I1Base {
    int a = 1000;
    void show();
    static void display() {  
        System.out.println("Hello from static method in I1Base");
    }       
}

interface I2 extends I1Base {
    void show2();
}

public class eI1 implements I2 {
    @Override
    public void show() {
        System.out.println("This is class eI1 (Implementing show). Value of a: " + a);
    }
    @Override
    public void show2() {
        System.out.println("This is show2 method (Implemented from I2)");
    }
    public static void main(String[] args) {
        System.out.println("--- Interface Inheritance ---");
        I1Base a1 = new eI1();
        a1.show();
        I1Base.display();
        I2 i2 = new eI1();
        i2.show2();
        i2.show(); // Calls the same implementation
    }
}

# 5.3 einterphase.java
// Demonstrates creating and implementing an interface. Shows that interface variables are static and final by default.
interface SimpleInterface {
    int a = 1000; // In interface, all variables are by default, public, static, and final (constant)
    void show(); // In interface, all methods are by default, abstract (no body)
    static void display() {
        System.out.println("Hello from static method in SimpleInterface");
    }
}

public class einterphase implements SimpleInterface {
    @Override
    public void show() {
        System.out.println("This is class einterphase implementing SimpleInterface. Value of a: " + a);
    }
    void displayLocal() {
        System.out.println("Hello from local method. Value of a: " + a);
    }
    public static void main(String[] args) {
        System.out.println("--- Interface Examples ---");
        SimpleInterface a1 = new einterphase();
        a1.show();
        SimpleInterface.display();
        einterphase obj = new einterphase();
        obj.displayLocal();
    }
}

# 6.2 fpackage.java
// Demonstrates the use of built-in packages like java.util.Scanner.
// Here java is a package, util is a subpackage, Scanner is a class of util package.
import java.util.Scanner;
public class fpackage {
    public static void main(String[] args) {
        try (Scanner sc = new Scanner(System.in)) {
            System.out.print("Enter first value: ");
            int a = sc.nextInt();
            System.out.print("Enter second value: ");
            int b = sc.nextInt();
            int sum = a + b;
            System.out.println("Sum Of the 2 number=" + sum);
        } finally {
            System.out.println("This block always runs.");
        }
    }
}
